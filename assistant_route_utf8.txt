# =========================================================
# ROUTER
# =========================================================

router = APIRouter(
    prefix="/assistant",
    tags=["Assistant"],
)


# =========================================================
# RUNTIME SERVICES
# =========================================================

_assistant_service: AssistantService | None = None

_conversation_service: ConversationService | None = None


# =========================================================
# ASSISTANT SERVICE CONFIGURATION
# =========================================================

def configure_assistant_service(
    service: AssistantService,
    conversation_service: ConversationService | None = None,
) -> None:
    """
    Inject các service cần thiết cho Assistant API.

    conversation_service là optional để bảo đảm tương thích
    với runtime cũ:

        configure_assistant_service(
            assistant_service
        )

    và runtime mới:

        configure_assistant_service(
            assistant_service,
            conversation_service=conversation_service,
        )
    """

    global _assistant_service
    global _conversation_service

    _assistant_service = service
    _conversation_service = conversation_service


def get_assistant_service() -> AssistantService:
    """
    Lấy AssistantService hiện tại.
    """

    if _assistant_service is None:
        raise HTTPException(
            status_code=(
                status.HTTP_503_SERVICE_UNAVAILABLE
            ),
            detail=(
                "Assistant Service chưa được "
                "khởi tạo trong API runtime."
            ),
        )

    return _assistant_service


# =========================================================
# CONVERSATION SERVICE CONFIGURATION
# =========================================================

def configure_conversation_service(
    service: ConversationService,
) -> None:
    """
    Inject ConversationService vào API runtime.

    Hàm này được giữ độc lập để tương thích với
    các runtime/test cũ có thể cấu hình service riêng.
    """

    global _conversation_service

    _conversation_service = service


def get_conversation_service() -> ConversationService:
    """
    Lấy ConversationService hiện tại.
    """

    if _conversation_service is None:
        raise HTTPException(
            status_code=(
                status.HTTP_503_SERVICE_UNAVAILABLE
            ),
            detail=(
                "Conversation Service chưa được "
                "khởi tạo trong API runtime."
            ),
        )

    return _conversation_service


# =========================================================
# REQUEST VALIDATION
# =========================================================

def _validate_question(
    request: AssistantRequest,
) -> str:
    """
    Validate và chuẩn hóa câu hỏi.

    Quy tắc:

        ""      → 400
        "   "   → 400
        None    → 422
        hợp lệ  → question.strip()
    """

    question: Any = request.question

    if not isinstance(question, str):
        raise HTTPException(
            status_code=(
                status.HTTP_422_UNPROCESSABLE_ENTITY
            ),
            detail="question phải là chuỗi.",
        )

    normalized = question.strip()

    if not normalized:
        raise HTTPException(
            status_code=(
                status.HTTP_400_BAD_REQUEST
            ),
            detail="Câu hỏi không được rỗng.",
        )

    return normalized


# =========================================================
# CITATION CONVERSION
# =========================================================

def _citation_to_schema(
    citation: Any,
) -> AssistantCitation:
    """
    Chuyển Citation domain model sang API schema.

    Không phụ thuộc cứng vào isinstance(Citation)
    để các object tương thích vẫn có thể được
    chuyển đổi tại API boundary.
    """

    # -----------------------------------------------------
    # Native Citation
    # -----------------------------------------------------

    if isinstance(citation, Citation):
        return AssistantCitation(
            citation_id=citation.citation_id,
            source=citation.source,
            score=citation.score,
            document_id=citation.document_id,
            page_number=citation.page_number,
            chunk_index=citation.chunk_index,
            content=citation.content,
            metadata=dict(citation.metadata),
            label=citation.label,
        )

    # -----------------------------------------------------
    # Generic compatible object
    # -----------------------------------------------------

    citation_id = getattr(
        citation,
        "citation_id",
        None,
    )

    source = getattr(
        citation,
        "source",
        None,
    )

    score = getattr(
        citation,
        "score",
        None,
    )

    document_id = getattr(
        citation,
        "document_id",
        None,
    )

    page_number = getattr(
        citation,
        "page_number",
        None,
    )

    chunk_index = getattr(
        citation,
        "chunk_index",
        None,
    )

    content = getattr(
        citation,
        "content",
        "",
    )

    metadata = getattr(
        citation,
        "metadata",
        {},
    )

    label = getattr(
        citation,
        "label",
        None,
    )

    # -----------------------------------------------------
    # Required fields
    # -----------------------------------------------------

    if citation_id is None:
        raise ValueError(
            "Citation thiếu citation_id."
        )

    if source is None:
        raise ValueError(
            "Citation thiếu source."
        )

    if score is None:
        raise ValueError(
            "Citation thiếu score."
        )

    # -----------------------------------------------------
    # Label fallback
    # -----------------------------------------------------

    if label is None:
        label_parts = [
            str(source)
        ]

        if page_number is not None:
            label_parts.append(
                f"trang {page_number}"
            )

        if chunk_index is not None:
            label_parts.append(
                f"chunk {chunk_index}"
            )

        label = " — ".join(
            label_parts
        )

    # -----------------------------------------------------
    # Metadata normalization
    # -----------------------------------------------------

    if not isinstance(
        metadata,
        dict,
    ):
        metadata = {}

    # -----------------------------------------------------
    # Schema
    # -----------------------------------------------------

    return AssistantCitation(
        citation_id=str(
            citation_id
        ),
        source=str(
            source
        ),
        score=float(
            score
        ),
        document_id=document_id,
        page_number=page_number,
        chunk_index=chunk_index,
        content=str(
            content or ""
        ),
        metadata=dict(
            metadata
        ),
        label=str(
            label
        ),
    )


# =========================================================
# CONVERSATION HELPERS
# =========================================================

def _get_conversation_if_requested(
    request: AssistantRequest,
) -> tuple[
    ConversationService | None,
    Any | None,
]:
    """
    Nếu request có conversation_id:

        1. Lấy ConversationService.
        2. Tìm Conversation.
        3. Nếu không tồn tại → HTTP 404.

    Nếu không có conversation_id:

        Không yêu cầu ConversationService.
    """

    conversation_id = getattr(
        request,
        "conversation_id",
        None,
    )

    if not conversation_id:
        return None, None

    conversation_service = (
        get_conversation_service()
    )

    try:
        conversation = (
            conversation_service.get(
                conversation_id
            )
        )

    except ValueError as exc:
        raise HTTPException(
            status_code=(
                status.HTTP_404_NOT_FOUND
            ),
            detail=(
                f"Không tìm thấy Conversation "
                f"'{conversation_id}'."
            ),
        ) from exc

    return (
        conversation_service,
        conversation,
    )


# =========================================================
# ASK
# =========================================================

@router.post(
    "/ask",
    response_model=AssistantResponseSchema,
    status_code=status.HTTP_200_OK,
    summary="Gửi câu hỏi cho AI Assistant",
)
def ask_assistant(
    request: AssistantRequest,
) -> AssistantResponseSchema:
    """
    Xử lý một câu hỏi tới AI Assistant.

    Nếu request có conversation_id:

        USER
          ↓
        Conversation
          ↓
        Assistant
          ↓
        ASSISTANT
          ↓
        Conversation

    Nhờ đó Conversation giữ được lịch sử
    cho các lượt chat tiếp theo.
    """

    # -----------------------------------------------------
    # 1. QUESTION
    # -----------------------------------------------------

    question = _validate_question(
        request
    )

    # -----------------------------------------------------
    # 2. CONVERSATION
    # -----------------------------------------------------

    (
        conversation_service,
        conversation,
    ) = _get_conversation_if_requested(
        request
    )

    # -----------------------------------------------------
    # 3. SAVE USER MESSAGE
    # -----------------------------------------------------

    if (
        conversation_service is not None
        and conversation is not None
    ):
        conversation_service.add_user_message(
            conversation.conversation_id,
            question,
        )

    # -----------------------------------------------------
    # 4. ASSISTANT
    # -----------------------------------------------------

    service = get_assistant_service()

    try:
        response = service.answer(
            question
        )

    except AssistantServiceError as exc:
        raise HTTPException(
            status_code=(
                status.HTTP_500_INTERNAL_SERVER_ERROR
            ),
            detail=str(exc),
        ) from exc

    except Exception as exc:
        raise HTTPException(
            status_code=(
                status.HTTP_500_INTERNAL_SERVER_ERROR
            ),
            detail=(
                "Đã xảy ra lỗi khi xử lý "
                "AI Assistant."
            ),
        ) from exc

    # -----------------------------------------------------
    # 5. CITATIONS
    # -----------------------------------------------------

    citations: list[
        AssistantCitation
    ] = []

    for citation in (
        response.citations or []
    ):
        try:
            citations.append(
                _citation_to_schema(
                    citation
                )
            )

        except Exception as exc:
            raise HTTPException(
                status_code=(
                    status.HTTP_500_INTERNAL_SERVER_ERROR
                ),
                detail=(
                    "Không thể chuyển đổi "
                    "Citation sang API response."
                ),
            ) from exc

    # -----------------------------------------------------
    # 6. METADATA
    # -----------------------------------------------------

    metadata = dict(
        response.metadata or {}
    )

    metadata[
        "citation_count"
    ] = len(citations)

    # -----------------------------------------------------
    # 7. SAVE ASSISTANT MESSAGE
    # -----------------------------------------------------

    if (
        conversation_service is not None
        and conversation is not None
    ):
        conversation_service.add_assistant_message(
            conversation.conversation_id,
            response.answer,
            provider=str(
                metadata.get(
                    "provider",
                    "",
                )
            ),
            model=str(
                metadata.get(
                    "model",
                    "",
                )
            ),
            tokens=int(
                metadata.get(
                    "tokens",
                    0,
                ) or 0
            ),
        )

        metadata[
            "conversation_id"
        ] = (
            conversation.conversation_id
        )

    # -----------------------------------------------------
    # 8. RESPONSE
    # -----------------------------------------------------

    return AssistantResponseSchema(
        success=True,
        question=response.query,
        answer=response.answer,
        citations=citations,
        metadata=metadata,
        message=(
            "Assistant xử lý câu hỏi "
            "thành công."
        ),
    )
